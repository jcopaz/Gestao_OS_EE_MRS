-- 1. TABELA DE USUÁRIOS
-- Utilizada para popular os dropdowns de técnicos e acompanhantes no modo offline.
CREATE TABLE IF NOT EXISTS usuarios (
    id SERIAL PRIMARY KEY,
    username VARCHAR(255) UNIQUE NOT NULL,
    perfil VARCHAR(100) -- Ex: 'Técnico', 'Planejador', 'Gestor'
    -- Outros campos adicionais como senha, base_padrao, etc., podem existir aqui.
);

-- 2. TABELA DE EVIDÊNCIAS (FOTOS)
-- Armazena as URLs do Supabase (ou Base64 em caso de fallback) atreladas ao Ativo/Atividade.
-- A chave primária composta garante o UPSERT via ON CONFLICT (ativo, atividade).
CREATE TABLE IF NOT EXISTS evidencias (
    ativo VARCHAR(100) NOT NULL,
    atividade VARCHAR(255) NOT NULL,
    foto_url TEXT,
    os_referencia VARCHAR(100),
    concluido_por VARCHAR(255),
    geolocalizacao TEXT,
    data_upload TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (ativo, atividade)
);

-- 3. TABELA DE BAIXAS (APONTAMENTOS DE OS)
-- Armazena os dados textuais, tempos e distâncias informados pelo técnico em campo.
-- A chave primária na OS garante o UPSERT via ON CONFLICT (os) evitando duplicidade.
CREATE TABLE IF NOT EXISTS baixas (
    os VARCHAR(100) PRIMARY KEY,
    status VARCHAR(100),
    realizado_em VARCHAR(100), -- Data/Hora da sincronização ou execução
    coordenacao VARCHAR(255),
    concluido_por VARCHAR(255),
    geolocalizacao_baixa TEXT,
    equipe TEXT,               -- Nome do acompanhante ou 'Sozinho'
    data_inicio VARCHAR(20),   -- Tratado para o SAP (Ex: recuo de madrugada)
    hora_inicio VARCHAR(20),
    data_fim VARCHAR(20),
    hora_fim VARCHAR(20),
    foto_evidencia TEXT        -- Base64 em caso de fallback, senão vazio/link
);
